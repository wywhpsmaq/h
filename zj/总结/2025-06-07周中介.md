2. **典型题目分析**：  
   - 1.[子树的重心](http://bbcoj.cn/training/17/problem/BC302)
     本题难了我好久，其实很简单。只需要 DFS后处理以每个节点为根的子树的重心。
```cpp
#include <bits/stdc++.h>
using namespace std;
int sz[1114514], ch[1114514], n, q, fa[1114514];
vector<int> t[1114514], cc[1114514];
void dfs(int u, int f) {
    sz[u] = 1;
    for (int v : t[u]) {
        if (v == f) continue;
        dfs(v, u);
        sz[u] += sz[v];
    }
}
void so(int u) {
    int c = u;
    while (true) {
        bool f = false;
        for (int v : t[c]) {
            if (v == fa[c]) continue;
            //这里加点什么，就可以AC了
            if (sz[v] > sz[u] / 2) {
                c = v;
                f = true;
                break;
            }
        }
        if (!f) break;
    }
    ch[u] = c;
    for (int v : cc[u]) {
        so(v);
    }
}
int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    cin >> n >> q;
    for (int i = 2; i <= n; ++i) {
        cin >> fa[i];
        t[fa[i]].push_back(i);
        t[i].push_back(fa[i]);
        cc[fa[i]].push_back(i);
    }
    memset(ch, -1, sizeof(ch));
    dfs(1, 0);
    so(1);
    while (q--) {
        int x;
        cin >> x;
        cout << ch[x] << '\n';
    }
    return 0;
}
```

  - 2[唯一的重心 ](http://bbcoj.cn/training/17/problem/BC303)
    本题难了我好久，其实很简单。只需要先看一下是否唯一，不唯一在删一条边使重心唯一。（虽然~~很难~~，但是答案不唯一，输出符合条件的答案均可！！！）。
  
```cpp
#include<bits/stdc++.h>
using namespace std;
vector<int> p[114514];
int sz[114514];
int n, t;
void dfs(int u, int fa) {
    sz[u] = 1;
    for (int v : p[u]) {
        if (v == fa) continue;
        dfs(v, u);
        sz[u] += sz[v];
    }
}
//看看这个函数有什么问题
int f(int u, int fa) {
    for (int v : p[u]) {
        if (v == fa) continue;
        if (sz[v] > n / 2) return f(v, u);
    }
    return u;
}
int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    cin >> t;
    while (t--) {
        cin >> n;
        for (int i = 1; i <= n; ++i) p[i].clear();
        vector<pair<int, int>> e;
        for (int i = 1; i < n; ++i) {
            int x, y;
            cin >> x >> y;
            p[x].push_back(y);
            p[y].push_back(x);
            e.push_back({x, y});
        }
        dfs(1, 0);
        int c = f(1, 0);
        int l = -1, pp = -1;
        for (int v : p[c]) {
            if (p[v].size() == 1) {
                l = v;
                pp = c;
                break;
            }
        }
        if (l == -1) {
            l = e[0].second;
            pp = e[0].first;
        }
        cout << pp << " " << l << '\n';
        for (int v : p[l]) {
            if (v != pp) {
                cout << l << " " << v << '\n';
                break;
            }
        }
        cout << pp << " " << l << '\n';
    }
    return 0;
}
```
